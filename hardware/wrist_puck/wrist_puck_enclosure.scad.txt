/*
 * =============================================================
 *  WRIST-PUCK DUAL-CHANNEL HAPTIC NODE — Enclosure
 * =============================================================
 *  Designed for: Seeed XIAO ESP32-C3, DRV2605L breakout,
 *                SG51R sub-micro servo, 10mm coin LRA,
 *                400-500 mAh LiPo, TP4056 charger, perfboard
 *
 *  Print settings: 0.2 mm layer, 20% infill, PLA or PETG
 *  No supports needed (designed for supportless printing)
 *
 *  Compile:  OpenSCAD → F6 (render) → Export STL
 *  Two parts: bottom shell + top lid (snap-fit)
 *
 *  Author: Copilot / Sean (Whitehorse, YT)
 *  License: MIT
 * =============================================================
 */

/* ── Render selector ─────────────────────────────────────────
 *  Set to "bottom", "lid", or "both" (exploded view)
 */
RENDER = "both";  // "bottom" | "lid" | "both"

/* ── Global parameters (mm) ────────────────────────────────── */

// Overall puck dimensions
PUCK_OUTER_D    = 68;      // outer diameter
PUCK_HEIGHT     = 22;      // total assembled height
WALL            = 2.0;     // shell wall thickness
FLOOR_T         = 1.6;     // bottom floor thickness
LID_T           = 1.6;     // lid thickness

// Snap-fit lip
LIP_H           = 2.5;     // engagement height
LIP_TOL         = 0.25;    // clearance per side

// Corner rounding
ROUND_R         = 3;

// ── Component cavities ──────────────────────────────────────

// XIAO ESP32-C3: 21 × 17.8 × 7 mm (with headers ~10mm tall)
XIAO_W   = 22;   // +1 clearance
XIAO_L   = 18.5; // +0.7 clearance
XIAO_H   = 10;   // with pin headers

// DRV2605L breakout: 18 × 17 × 3 mm (Adafruit style)
DRV_W    = 19;
DRV_L    = 18;
DRV_H    = 5;    // with soldered headers

// SG51R servo: 25 × 22 × 12 mm
SERVO_W  = 26;   // +1
SERVO_L  = 23;   // +1
SERVO_H  = 13;   // +1

// Sliding mass rail slot (runs alongside servo)
RAIL_W   = 6;
RAIL_L   = 30;
RAIL_H   = 5;

// 10mm coin LRA: Ø10 × 3.7 mm
LRA_D    = 11;   // +1
LRA_H    = 4.5;  // +0.8

// LiPo battery: ~35 × 25 × 5 mm (typ. 400-500 mAh)
BAT_W    = 36;   // +1
BAT_L    = 26;   // +1
BAT_H    = 6;    // +1

// TP4056 charger module: ~26 × 18 × 4 mm
CHG_W    = 27;
CHG_L    = 19;
CHG_H    = 5;

// Mini perfboard: 20 × 15 × 1.6 mm
PERF_W   = 21;
PERF_L   = 16;
PERF_H   = 8;    // with components soldered

// Strap slot dimensions
STRAP_W  = 22;   // watchband width
STRAP_H  = 3;
STRAP_D  = 5;    // depth into wall

// USB-C port cutout
USBC_W   = 10;
USBC_H   = 4;

$fn = 80;

/* ── Derived ─────────────────────────────────────────────────── */
INNER_D    = PUCK_OUTER_D - 2*WALL;
INNER_R    = INNER_D / 2;
OUTER_R    = PUCK_OUTER_D / 2;
CAVITY_H   = PUCK_HEIGHT - FLOOR_T - LID_T;

/* ════════════════════════ MODULES ════════════════════════════ */

// Rounded cylinder (outer shell shape)
module rounded_cylinder(d, h, r) {
  hull() {
    translate([0, 0, r])
      cylinder(d=d-2*r, h=h-2*r);
    translate([0, 0, r])
      rotate_extrude()
        translate([(d/2)-r, 0])
          circle(r=r);
    translate([0, 0, h-r])
      rotate_extrude()
        translate([(d/2)-r, 0])
          circle(r=r);
  }
}

// Rectangular pocket (centered at origin, extends downward from z=0)
module pocket(w, l, h) {
  translate([-w/2, -l/2, 0])
    cube([w, l, h]);
}

// Mounting post (screw boss)
module post(d_outer, d_hole, h) {
  difference() {
    cylinder(d=d_outer, h=h);
    translate([0, 0, -0.1])
      cylinder(d=d_hole, h=h+0.2);
  }
}

/* ────────────────── BOTTOM SHELL ────────────────────────────── */
module bottom_shell() {
  difference() {
    union() {
      // Outer shell
      difference() {
        // Main body
        cylinder(d=PUCK_OUTER_D, h=PUCK_HEIGHT - LID_T);
        
        // Hollow interior
        translate([0, 0, FLOOR_T])
          cylinder(d=INNER_D, h=CAVITY_H + 1);
        
        // ── Strap slots (12 o'clock and 6 o'clock) ──
        // Top strap slot
        translate([-STRAP_W/2, OUTER_R - STRAP_D, FLOOR_T])
          cube([STRAP_W, STRAP_D + 1, STRAP_H]);
        // Bottom strap slot
        translate([-STRAP_W/2, -OUTER_R - 1, FLOOR_T])
          cube([STRAP_W, STRAP_D + 1, STRAP_H]);
      }
      
      // ── Snap-fit inner lip (for lid engagement) ──
      translate([0, 0, PUCK_HEIGHT - LID_T - LIP_H])
        difference() {
          cylinder(d=INNER_D, h=LIP_H);
          cylinder(d=INNER_D - 2*1.2, h=LIP_H + 0.1);
        }
      
      // ── Component mounting features ──
      
      // M2 mounting posts (4×, near corners of interior)
      for (a = [45, 135, 225, 315]) {
        translate([cos(a) * (INNER_R - 5), sin(a) * (INNER_R - 5), FLOOR_T])
          post(5, 1.6, CAVITY_H - 2);
      }
    }
    
    // ══════ COMPONENT CAVITIES ══════
    
    // ── LRA face pocket (top-center of lid face, recessed into floor) ──
    // The LRA sits in a shallow pocket in the floor so vibration
    // transmits directly to the skin side
    translate([0, INNER_R/2, -0.1])
      cylinder(d=LRA_D, h=LRA_H + 0.1);
    // LRA wire channel
    translate([-1.5, INNER_R/2 - LRA_D/2 - 1, 0])
      cube([3, LRA_D/2 + 2, LRA_H]);
    
    // ── Servo cavity (left side, oriented lengthwise) ──
    translate([-INNER_R/2 + 2, -SERVO_L/2 + 2, FLOOR_T])
      pocket(SERVO_W, SERVO_L, SERVO_H);
    // Servo horn slot (extends upward)
    translate([-INNER_R/2 + 2, -SERVO_L/2 + 2 + SERVO_L/2, FLOOR_T + SERVO_H - 1])
      cylinder(d=12, h=5);
    
    // ── Sliding mass rail slot (alongside servo) ──
    translate([-INNER_R/2 + 2 + SERVO_W/2 + RAIL_W/2 + 2, -RAIL_L/2, FLOOR_T])
      pocket(RAIL_W, RAIL_L, RAIL_H);
    
    // ── XIAO ESP32-C3 pocket (right side) ──
    translate([INNER_R/2 - 4, -XIAO_L/2, FLOOR_T])
      pocket(XIAO_W, XIAO_L, XIAO_H);
    
    // ── USB-C port cutout (right wall, aligned with XIAO) ──
    translate([OUTER_R - WALL - 1, -USBC_W/2, FLOOR_T + 2])
      cube([WALL + 2, USBC_W, USBC_H]);
    
    // ── DRV2605L breakout pocket (front-right) ──
    translate([INNER_R/4, -INNER_R/2 + 2, FLOOR_T])
      pocket(DRV_W, DRV_L, DRV_H);
    
    // ── Battery compartment (rear center) ──
    translate([0, -INNER_R/2 + BAT_L/2 - 2, FLOOR_T])
      pocket(BAT_W, BAT_L, BAT_H);
    
    // ── TP4056 charger pocket (rear, below battery) ──
    translate([0, -INNER_R + 6, FLOOR_T])
      pocket(CHG_W, CHG_L, CHG_H);
    // USB-C charge port cutout (rear wall)
    translate([-USBC_W/2, -OUTER_R - 1, FLOOR_T + 1])
      cube([USBC_W, WALL + 2, USBC_H]);
    
    // ── Perfboard pocket (above battery) ──
    translate([0, 0, FLOOR_T + BAT_H + 0.5])
      pocket(PERF_W, PERF_L, PERF_H);
    
    // ── Wire routing channels (floor-level grooves) ──
    // I2C bus channel (XIAO → DRV2605L)
    translate([INNER_R/4 - 1, -2, FLOOR_T - 0.5])
      cube([INNER_R/4 + 4, 4, 2]);
    // Servo signal channel (XIAO → servo)
    translate([-INNER_R/2 + 4, -2, FLOOR_T - 0.5])
      cube([INNER_R - 4, 4, 2]);
    // LRA wire channel (floor to DRV2605L)
    translate([-1, 0, FLOOR_T - 0.5])
      cube([2, INNER_R/2 + 5, 2]);
    
    // ── Ventilation holes (bottom, small grid) ──
    for (a = [0 : 60 : 300]) {
      translate([cos(a) * 12, sin(a) * 12, -0.1])
        cylinder(d=2, h=FLOOR_T + 0.2);
    }
  }
}

/* ────────────────── LID ─────────────────────────────────────── */
module lid() {
  difference() {
    union() {
      // Main lid disc
      cylinder(d=PUCK_OUTER_D, h=LID_T);
      
      // Snap-fit engagement ring (inner, drops down)
      translate([0, 0, -LIP_H])
        difference() {
          cylinder(d=INNER_D - 2*LIP_TOL, h=LIP_H);
          cylinder(d=INNER_D - 2*1.2 - 2*LIP_TOL, h=LIP_H + 0.1);
        }
    }
    
    // ── LRA window (thin membrane or open for direct skin contact) ──
    // Option A: open hole (strongest haptic)
    translate([0, INNER_R/2, -LIP_H - 0.1])
      cylinder(d=LRA_D - 1, h=LID_T + LIP_H + 0.2);
    
    // ── Decorative top pattern (subtle rings) ──
    for (r = [10, 20, 30]) {
      difference() {
        translate([0, 0, LID_T - 0.3])
          cylinder(d=r*2, h=0.4);
        translate([0, 0, LID_T - 0.4])
          cylinder(d=r*2 - 1.2, h=0.6);
      }
    }
    
    // ── "HP" text label (Haptic Puck) ──
    translate([0, -8, LID_T - 0.4])
      linear_extrude(0.5)
        text("HP", size=8, halign="center", valign="center",
             font="Liberation Sans:style=Bold");
    
    // ── Status LED window (small hole) ──
    translate([10, -INNER_R/2 + 3, -LIP_H - 0.1])
      cylinder(d=3, h=LID_T + LIP_H + 0.2);
    
    // ── Strap clearance notches ──
    translate([-STRAP_W/2, OUTER_R - STRAP_D, -LIP_H - 0.1])
      cube([STRAP_W, STRAP_D + 1, STRAP_H + 0.1]);
    translate([-STRAP_W/2, -OUTER_R - 1, -LIP_H - 0.1])
      cube([STRAP_W, STRAP_D + 1, STRAP_H + 0.1]);
  }
}

/* ════════════════════════ RENDER ═════════════════════════════ */

if (RENDER == "bottom") {
  bottom_shell();
}
else if (RENDER == "lid") {
  // Flip lid for printing (flat side down)
  translate([0, 0, LID_T])
    rotate([180, 0, 0])
      lid();
}
else if (RENDER == "both") {
  // Exploded view
  color("SteelBlue", 0.85)
    bottom_shell();
  
  color("LightSkyBlue", 0.7)
    translate([0, 0, PUCK_HEIGHT + 8])  // exploded offset
      lid();
}
